-- Controlled CPU Stress Test for Roblox (SAFE VERSION)

local RunService = game:GetService("RunService")

local stressEnabled = true      -- toggle switch
local intensity = 500000          -- increase = more CPU usage (try 1000 - 20000)

-- optional: stop after some time (prevents freezing forever)
local maxSeconds = 500000
local startTime = tick()

local function heavyComputation()
    local x = 0
    for i = 1, intensity do
        x = math.sin(i) * math.cos(i) * math.sqrt(i)
    end
    return x
end

RunService.Heartbeat:Connect(function()
    if not stressEnabled then return end

    -- auto stop safety
    if tick() - startTime > maxSeconds then
        stressEnabled = false
        warn("Stress test ended automatically (safety limit reached)")
        return
    end

    -- do CPU-heavy work
    local result = heavyComputation()

    -- prevent optimization removal
    if result == math.huge then
        print("impossible")
    end
end)
